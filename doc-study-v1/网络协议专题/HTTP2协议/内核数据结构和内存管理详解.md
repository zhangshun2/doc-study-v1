### 1.2 sk_buff内存布局
```
sk_buff结构体内存布局：

+-------------------+
|   sk_buff 结构    |  <- 控制信息
+-------------------+
|                   |
|   head            |  <- 指向数据缓冲区开始
|   data            |  <- 指向有效数据开始  
|   tail            |  <- 指向有效数据结束
|   end             |  <- 指向数据缓冲区结束
|                   |
+-------------------+
|                   |
|   数据缓冲区       |  <- 实际网络数据
|                   |
+-------------------+
| skb_shared_info   |  <- 分片信息
+-------------------+

内存指针关系：
head <= data <= tail <= end

可用空间：
- headroom: data - head  (用于添加头部)
- tailroom: end - tail   (用于添加尾部)
- 数据长度: tail - data
```

## 3. 内存分配和管理

### 3.1 sk_buff分配
```c
/* 分配sk_buff */
struct sk_buff *alloc_skb(unsigned int size, gfp_t priority) {
    struct sk_buff *skb;
    u8 *data;
    
    /* 分配sk_buff结构 */
    skb = kmem_cache_alloc(skbuff_head_cache, priority);
    if (unlikely(!skb))
        return NULL;
    
    /* 计算实际需要的大小 */
    size = SKB_DATA_ALIGN(size);
    size += SKB_DATA_ALIGN(sizeof(struct skb_shared_info));
    
    /* 分配数据缓冲区 */
    data = kmalloc(size, priority);
    if (unlikely(!data)) {
        kmem_cache_free(skbuff_head_cache, skb);
        return NULL;
    }
    
    /* 初始化sk_buff */
    memset(skb, 0, offsetof(struct sk_buff, tail));
    skb->truesize = SKB_TRUESIZE(size);
    atomic_set(&skb->users, 1);
    skb->head = data;
    skb->data = data;
    skb_reset_tail_pointer(skb);
    skb->end = skb->tail + size;
    skb->mac_header = (typeof(skb->mac_header))~0U;
    skb->transport_header = (typeof(skb->transport_header))~0U;
    
    /* 初始化shared_info */
    shinfo = skb_shinfo(skb);
    memset(shinfo, 0, offsetof(struct skb_shared_info, dataref));
    atomic_set(&shinfo->dataref, 1);
    kmemcheck_annotate_variable(shinfo->destructor_arg);
    
    return skb;
}

/* 释放sk_buff */
void kfree_skb(struct sk_buff *skb) {
    if (unlikely(!skb))
        return;
    
    if (likely(atomic_read(&skb->users) == 1))
        smp_rmb();
    else if (likely(!atomic_dec_and_test(&skb->users)))
        return;
    
    trace_kfree_skb(skb, __builtin_return_address(0));
    __kfree_skb(skb);
}

void __kfree_skb(struct sk_buff *skb) {
    dst_release(skb_dst(skb));
    secpath_put(skb->sp);
    
    if (skb->destructor) {
        WARN_ON(in_irq());
        skb->destructor(skb);
    }
    
    nf_conntrack_put(skb->nfct);
    nf_bridge_put(skb->nf_bridge);
    
    skb_release_head_state(skb);
    skb_release_data(skb);
}
```

### 3.2 内存池管理
```c
/* sk_buff内存池 */
static struct kmem_cache *skbuff_head_cache __read_mostly;
static struct kmem_cache *skbuff_fclone_cache __read_mostly;

/* 初始化内存池 */
void __init skb_init(void) {
    skbuff_head_cache = kmem_cache_create("skbuff_head_cache",
                                         sizeof(struct sk_buff),
                                         0,
                                         SLAB_HWCACHE_ALIGN|SLAB_PANIC,
                                         NULL);
    
    skbuff_fclone_cache = kmem_cache_create("skbuff_fclone_cache",
                                           2 * sizeof(struct sk_buff),
                                           0,
                                           SLAB_HWCACHE_ALIGN|SLAB_PANIC,
                                           NULL);
}

/* 快速分配（从内存池） */
struct sk_buff *__alloc_skb(unsigned int size, gfp_t gfp_mask,
                           int flags, int node) {
    struct kmem_cache *cache;
    struct skb_shared_info *shinfo;
    struct sk_buff *skb;
    u8 *data;
    bool pfmemalloc;
    
    cache = (flags & SKB_ALLOC_FCLONE)
        ? skbuff_fclone_cache : skbuff_head_cache;
    
    if (sk_memalloc_socks() && (flags & SKB_ALLOC_RX))
        gfp_mask |= __GFP_MEMALLOC;
    
    /* 从内存池分配 */
    skb = kmem_cache_alloc_node(cache, gfp_mask & ~__GFP_DMA, node);
    if (unlikely(!skb))
        return NULL;
    
    prefetchw(skb);
    
    /* 分配数据缓冲区 */
    size = SKB_DATA_ALIGN(size);
    size += SKB_DATA_ALIGN(sizeof(struct skb_shared_info));
    data = kmalloc_reserve(size, gfp_mask, node, &pfmemalloc);
    if (unlikely(!data))
        goto nodata;
    
    /* 预取数据缓冲区 */
    prefetchw(data + size);
    
    /* 初始化sk_buff */
    memset(skb, 0, offsetof(struct sk_buff, tail));
    skb->truesize = SKB_TRUESIZE(size);
    skb->pfmemalloc = pfmemalloc;
    atomic_set(&skb->users, 1);
    skb->head = data;
    skb->data = data;
    skb_reset_tail_pointer(skb);
    skb->end = skb->tail + size;
    skb->mac_header = (typeof(skb->mac_header))~0U;
    skb->transport_header = (typeof(skb->transport_header))~0U;
    
    /* 初始化shared_info */
    shinfo = skb_shinfo(skb);
    memset(shinfo, 0, offsetof(struct skb_shared_info, dataref));
    atomic_set(&shinfo->dataref, 1);
    
    if (flags & SKB_ALLOC_FCLONE) {
        struct sk_buff_fclones *fclones;
        
        fclones = container_of(skb, struct sk_buff_fclones, skb1);
        
        kmemcheck_annotate_bitfield(&fclones->skb2, flags1);
        skb->fclone = SKB_FCLONE_ORIG;
        atomic_set(&fclones->fclone_ref, 1);
        
        fclones->skb2.fclone = SKB_FCLONE_CLONE;
        fclones->skb2.pfmemalloc = pfmemalloc;
    }
    
    return skb;
    
nodata:
    kmem_cache_free(cache, skb);
    return NULL;
}
```

## 4. 队列管理

### 4.1 sk_buff_head队列结构
```c
struct sk_buff_head {
    /* 队列链表 */
    struct sk_buff *next;
    struct sk_buff *prev;
    
    /* 队列长度 */
    __u32 qlen;
    
    /* 队列锁 */
    spinlock_t lock;
};

/* 队列初始化 */
static inline void skb_queue_head_init(struct sk_buff_head *list) {
    spin_lock_init(&list->lock);
    __skb_queue_head_init(list);
}

static inline void __skb_queue_head_init(struct sk_buff_head *list) {
    list->prev = list->next = (struct sk_buff *)list;
    list->qlen = 0;
}
```

### 4.2 队列操作函数
```c
/* 在队列尾部添加skb */
static inline void skb_queue_tail(struct sk_buff_head *list,
                                 struct sk_buff *newsk) {
    unsigned long flags;
    
    spin_lock_irqsave(&list->lock, flags);
    __skb_queue_tail(list, newsk);
    spin_unlock_irqrestore(&list->lock, flags);
}

static inline void __skb_queue_tail(struct sk_buff_head *list,
                                   struct sk_buff *newsk) {
    __skb_queue_before(list, (struct sk_buff *)list, newsk);
}

static inline void __skb_queue_before(struct sk_buff_head *list,
                                     struct sk_buff *next,
                                     struct sk_buff *newsk) {
    __skb_insert(newsk, next->prev, next, list);
}

static inline void __skb_insert(struct sk_buff *newsk,
                               struct sk_buff *prev, struct sk_buff *next,
                               struct sk_buff_head *list) {
    newsk->next = next;
    newsk->prev = prev;
    next->prev  = prev->next = newsk;
    list->qlen++;
}

/* 从队列头部取出skb */
static inline struct sk_buff *skb_dequeue(struct sk_buff_head *list) {
    unsigned long flags;
    struct sk_buff *result;
    
    spin_lock_irqsave(&list->lock, flags);
    result = __skb_dequeue(list);
    spin_unlock_irqrestore(&list->lock, flags);
    return result;
}

static inline struct sk_buff *__skb_dequeue(struct sk_buff_head *list) {
    struct sk_buff *skb = skb_peek(list);
    if (skb)
        __skb_unlink(skb, list);
    return skb;
}

static inline void __skb_unlink(struct sk_buff *skb, struct sk_buff_head *list) {
    struct sk_buff *next, *prev;
    
    list->qlen--;
    next       = skb->next;
    prev       = skb->prev;
    skb->next  = skb->prev = NULL;
    next->prev = prev;
    prev->next = next;
}
```

## 5. 内存拷贝机制

### 5.1 用户空间到内核空间拷贝
```c
/* 安全的数据拷贝 */
unsigned long copy_from_user(void *to, const void __user *from, unsigned long n) {
    if (access_ok(VERIFY_READ, from, n))
        return __copy_from_user(to, from, n);
    return n;
}

unsigned long copy_to_user(void __user *to, const void *from, unsigned long n) {
    if (access_ok(VERIFY_WRITE, to, n))
        return __copy_to_user(to, from, n);
    return n;
}

/* 检查用户空间地址有效性 */
static inline int access_ok(int type, const void __user *addr, unsigned long size) {
    unsigned long limit = current_thread_info()->addr_limit.seg;
    
    return likely(((unsigned long)addr + size) <= limit);
}

/* 实际的拷贝函数（架构相关） */
static inline unsigned long __copy_from_user(void *to, const void __user *from, unsigned long n) {
    might_fault();
    return __copy_from_user_inatomic(to, from, n);
}

/* x86_64架构的优化拷贝 */
static __always_inline unsigned long __copy_from_user_inatomic(void *to, const void __user *from, unsigned long size) {
    return copy_user_generic(to, (__force void *)from, size);
}
```

### 5.2 sk_buff数据拷贝
```c
/* 拷贝数据到sk_buff */
int skb_copy_datagram_from_iter(struct sk_buff *skb, int offset,
                               struct iov_iter *from, int len) {
    int start = skb_headlen(skb);
    int i, copy = start - offset;
    struct sk_buff *frag_iter;
    
    /* 拷贝到线性数据区 */
    if (copy > 0) {
        if (copy > len)
            copy = len;
        if (copy_from_iter(skb->data + offset, copy, from) != copy)
            goto fault;
        if ((len -= copy) == 0)
            return 0;
        offset += copy;
    }
    
    /* 拷贝到分片 */
    for (i = 0; i < skb_shinfo(skb)->nr_frags; i++) {
        int end;
        const skb_frag_t *frag = &skb_shinfo(skb)->frags[i];
        
        WARN_ON(start > offset + len);
        
        end = start + skb_frag_size(frag);
        if ((copy = end - offset) > 0) {
            size_t copied;
            
            if (copy > len)
                copy = len;
            
            copied = copy_page_from_iter(skb_frag_page(frag),
                                       frag->page_offset + offset - start,
                                       copy, from);
            if (copied != copy)
                goto fault;
            
            if (!(len -= copy))
                return 0;
            offset += copy;
        }
        start = end;
    }
    
    /* 拷贝到分片链表 */
    skb_walk_frags(skb, frag_iter) {
        int end;
        
        WARN_ON(start > offset + len);
        
        end = start + frag_iter->len;
        if ((copy = end - offset) > 0) {
            if (copy > len)
                copy = len;
            if (skb_copy_datagram_from_iter(frag_iter,
                                          offset - start,
                                          from, copy))
                goto fault;
            if ((len -= copy) == 0)
                return 0;
            offset += copy;
        }
        start = end;
    }
    
    if (!len)
        return 0;
    
fault:
    return -EFAULT;
}

/* 从sk_buff拷贝数据到用户空间 */
int skb_copy_datagram_msg(const struct sk_buff *skb, int offset,
                         struct msghdr *msg, int size) {
    int start = skb_headlen(skb);
    int i, copy = start - offset;
    struct sk_buff *frag_iter;
    
    /* 从线性数据区拷贝 */
    if (copy > 0) {
        if (copy > size)
            copy = size;
        if (copy_to_iter(skb->data + offset, copy, &msg->msg_iter) != copy)
            goto fault;
        if (!(size -= copy))
            return 0;
        offset += copy;
    }
    
    /* 从分片拷贝 */
    for (i = 0; i < skb_shinfo(skb)->nr_frags; i++) {
        int end;
        const skb_frag_t *frag = &skb_shinfo(skb)->frags[i];
        
        WARN_ON(start > offset + size);
        
        end = start + skb_frag_size(frag);
        if ((copy = end - offset) > 0) {
            if (copy > size)
                copy = size;
            
            if (copy_page_to_iter(skb_frag_page(frag),
                                frag->page_offset + offset - start,
                                copy, &msg->msg_iter) != copy)
                goto fault;
            
            if (!(size -= copy))
                return 0;
            offset += copy;
        }
        start = end;
    }
    
    /* 从分片链表拷贝 */
    skb_walk_frags(skb, frag_iter) {
        int end;
        
        WARN_ON(start > offset + size);
        
        end = start + frag_iter->len;
        if ((copy = end - offset) > 0) {
            if (copy > size)
                copy = size;
            if (skb_copy_datagram_msg(frag_iter, offset - start,
                                    msg, copy))
                goto fault;
            if (!(size -= copy))
                return 0;
            offset += copy;
        }
        start = end;
    }
    
    if (!size)
        return 0;
    
fault:
    return -EFAULT;
}
```

## 6. 缓冲区管理

### 6.1 Socket缓冲区限制
```c
/* 检查发送缓冲区是否有空间 */
static inline bool sk_stream_memory_free(const struct sock *sk) {
    if (sk->sk_wmem_queued >= sk->sk_sndbuf)
        return false;
    
    return sk->sk_wmem_queued < sk->sk_sndbuf / 2 ||
           sk->sk_wmem_queued < sk->sk_sndbuf &&
           sk->sk_wmem_queued < sk->sk_sndbuf - atomic_read(&sk->sk_wmem_alloc);
}

/* 等待发送缓冲区空间 */
int sk_stream_wait_memory(struct sock *sk, long *timeo_p) {
    int err = 0;
    long vm_wait = 0;
    long current_timeo = *timeo_p;
    DEFINE_WAIT(wait);
    
    if (sk_stream_memory_free(sk))
        current_timeo = vm_wait = (prandom_u32() % (HZ / 5)) + 2;
    
    while (1) {
        set_bit(SOCK_ASYNC_NOSPACE, &sk->sk_socket->flags);
        
        prepare_to_wait(sk_sleep(sk), &wait, TASK_INTERRUPTIBLE);
        
        if (sk->sk_err || (sk->sk_shutdown & SEND_SHUTDOWN))
            goto do_error;
        
        if (!*timeo_p)
            goto do_nonblock;
        
        if (signal_pending(current))
            goto do_interrupted;
        
        clear_bit(SOCK_ASYNC_NOSPACE, &sk->sk_socket->flags);
        
        if (sk_stream_memory_free(sk) && !vm_wait)
            break;
        
        set_bit(SOCK_NOSPACE, &sk->sk_socket->flags);
        sk->sk_write_pending++;
        sk_wait_event(sk, &current_timeo, sk->sk_err ||
                     (sk->sk_shutdown & SEND_SHUTDOWN) ||
                     (sk_stream_memory_free(sk) && !vm_wait));
        sk->sk_write_pending--;
        
        if (vm_wait) {
            vm_wait -= current_timeo;
            current_timeo = *timeo_p;
            if (current_timeo != MAX_SCHEDULE_TIMEOUT &&
                (current_timeo -= vm_wait) < 0)
                current_timeo = 0;
            vm_wait = 0;
        }
        *timeo_p = current_timeo;
    }
out:
    finish_wait(sk_sleep(sk), &wait);
    return err;
    
do_error:
    err = -EPIPE;
    goto out;
do_nonblock:
    err = -EAGAIN;
    goto out;
do_interrupted:
    err = sock_intr_errno(*timeo_p);
    goto out;
}
```

```c
/* 更新Socket内存统计 */
static inline void sk_mem_charge(struct sock *sk, int size) {
    if (!sk_has_account(sk))
        return;
    sk->sk_forward_alloc -= size;
}

static inline void sk_mem_uncharge(struct sock *sk, int size) {
    if (!sk_has_account(sk))
        return;
    sk->sk_forward_alloc += size;
}

/* 检查内存限制 */
static inline bool sk_under_memory_pressure(const struct sock *sk) {
    if (mem_cgroup_sockets_enabled && sk->sk_cgrp)
        return !!sk->sk_cgrp->memory_pressure;
    
    return !!sk->sk_prot->memory_pressure;
}

/* 内存回收 */
static inline void sk_mem_reclaim(struct sock *sk) {
    if (!sk_has_account(sk))
        return;
    if (sk->sk_forward_alloc >= SK_MEM_QUANTUM)
        __sk_mem_reclaim(sk);
}

void __sk_mem_reclaim(struct sock *sk) {
    sk->sk_prot->memory_allocated -= sk->sk_forward_alloc >> SK_MEM_QUANTUM_SHIFT;
    sk->sk_forward_alloc &= SK_MEM_QUANTUM - 1;
    
    if (sk->sk_prot->memory_pressure && sk->sk_prot->memory_allocated < sk->sk_prot->sysctl_mem[0])
        *sk->sk_prot->memory_pressure = 0;
}
```

### 7.1 零拷贝技术
```c
/* sendfile系统调用 - 零拷贝文件传输 */
ssize_t do_sendfile(int out_fd, int in_fd, loff_t *ppos, size_t count, loff_t max) {
    struct fd in, out;
    struct inode *in_inode, *out_inode;
    loff_t pos;
    loff_t out_pos;
    ssize_t retval;
    int fl;
    
    /* 获取文件描述符 */
    retval = -EBADF;
    in = fdget(in_fd);
    if (!in.file)
        goto out;
    
    if (!(in.file->f_mode & FMODE_READ))
        goto fput_in;
    
    retval = -ESPIPE;
    if (!ppos) {
        pos = in.file->f_pos;
        ppos = &pos;
    } else {
        if (!(in.file->f_mode & FMODE_PREAD))
            goto fput_in;
    }
    retval = rw_verify_area(READ, in.file, ppos, count);
    if (retval < 0)
        goto fput_in;
    count = retval;
    
    retval = -EBADF;
    out = fdget(out_fd);
    if (!out.file)
        goto fput_in;
    
    if (!(out.file->f_mode & FMODE_WRITE))
        goto fput_out;
    
    retval = -EINVAL;
    in_inode = file_inode(in.file);
    out_inode = file_inode(out.file);
    out_pos = out.file->f_pos;
    retval = rw_verify_area(WRITE, out.file, &out_pos, count);
    if (retval < 0)
        goto fput_out;
    count = retval;
    
    if (!max)
        max = min(in_inode->i_sb->s_maxbytes, out_inode->i_sb->s_maxbytes);
    
    if (unlikely(*ppos + count > max)) {
        retval = -EOVERFLOW;
        goto fput_out;
    }
    
    fl = 0;
    if (in.file->f_flags & O_NONBLOCK)
        fl = SPLICE_F_NONBLOCK;
    
    file_start_write(out.file);
    retval = do_splice_direct(in.file, ppos, out.file, &out_pos, count, fl);
    file_end_write(out.file);
    
    if (retval > 0) {
        add_rchar(current, retval);
        add_wchar(current, retval);
        fsnotify_access(in.file);
        fsnotify_modify(out.file);
        out.file->f_pos = out_pos;
        if (ppos == &pos)
            in.file->f_pos = pos;
    }
    
    inc_syscr(current);
    inc_syscw(current);
    if (*ppos > max)
        retval = -EOVERFLOW;
    
fput_out:
    fdput(out);
fput_in:
    fdput(in);
out:
    return retval;
}
```

### 7.2 内存预分配
```c
/* 预分配sk_buff */
struct sk_buff *sk_stream_alloc_skb(struct sock *sk, int size, gfp_t gfp) {
    struct sk_buff *skb;
    
    /* 检查内存限制 */
    if (likely(sk_wmem_schedule(sk, size))) {
        skb = alloc_skb_fclone(size, gfp);
        if (likely(skb)) {
            skb_reserve(skb, sk_stream_alloc_skb_reserve(sk, size, skb));
            return skb;
        }
        sk->sk_prot->enter_memory_pressure(sk);
        sk_stream_moderate_sndbuf(sk);
    }
    return NULL;
}

/* 内存调度检查 */
static inline bool sk_wmem_schedule(struct sock *sk, int size) {
    if (!sk_has_account(sk))
        return true;
    return size <= sk->sk_forward_alloc ||
           __sk_mem_schedule(sk, size, SK_MEM_SEND);
}

int __sk_mem_schedule(struct sock *sk, int size, int kind) {
    struct proto *prot = sk->sk_prot;
    int amt = sk_mem_pages(size);
    long allocated;
    
    sk->sk_forward_alloc += amt << SK_MEM_QUANTUM_SHIFT;
    
    allocated = sk_memory_allocated_add(sk, amt);
    
    if (mem_cgroup_sockets_enabled && sk->sk_cgrp &&
        !mem_cgroup_charge_skmem(sk->sk_cgrp, amt))
        goto suppress_allocation;
    
    /* 检查全局内存限制 */
    if (allocated <= prot->sysctl_mem[0]) {
        sk_leave_memory_pressure(sk);
        return 1;
    }
    
    /* 进入内存压力状态 */
    sk_enter_memory_pressure(sk);
    
    /* 检查是否超过硬限制 */
    if (allocated > prot->sysctl_mem[2])
        goto suppress_allocation;
    
    /* 在软限制和硬限制之间，允许但记录 */
    if (allocated > prot->sysctl_mem[1])
        sk_memory_allocated_sub(sk, amt);
    
    return 1;
    
suppress_allocation:
    if (kind == SK_MEM_SEND && sk->sk_type == SOCK_STREAM) {
        sk_stream_moderate_sndbuf(sk);
        
        /* 失败时回滚 */
        sk->sk_forward_alloc -= amt << SK_MEM_QUANTUM_SHIFT;
        sk_memory_allocated_sub(sk, amt);
        return 0;
    }
    
    sk_memory_allocated_sub(sk, amt);
    return 0;
}
```
