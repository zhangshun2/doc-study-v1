# 状态栈结构原理详解
## 📚 状态栈的基本概念
### 什么是状态栈？
状态栈（State Stack）是一种**专门用于管理程序执行状态的栈结构**。它将程序的不同执行阶段、上下文信息或状态转换过程保存在栈中，支持状态的保存、恢复、回溯和切换。

### 状态栈的核心特征
```plain
状态栈的层次结构：

栈顶 → 当前状态     [State 3] ← 当前活跃状态
      ↓
      父状态       [State 2] ← 上一级状态  
      ↓
      根状态       [State 1] ← 初始状态
栈底 ↑

特点：
✓ 状态继承：子状态可以访问父状态的信息
✓ 状态隔离：每个状态有独立的作用域
✓ 状态回溯：可以回退到之前的状态
✓ 状态切换：支持状态间的平滑转换
```

### 状态栈的应用场景
1. **游戏状态管理**：菜单、游戏、暂停等状态切换
2. **编译器设计**：语法分析中的状态机
3. **用户界面**：页面导航和状态管理
4. **事务处理**：数据库事务的嵌套和回滚
5. **算法实现**：回溯算法、状态空间搜索

### 1. 基础状态栈框架
```java
/**
 * 通用状态栈实现
 */
public class StateStack<T extends State> {
    
    private final Deque<T> stateStack;
    private final StateFactory<T> stateFactory;
    private final StateTransitionValidator validator;
    
    public StateStack(StateFactory<T> stateFactory) {
        this.stateStack = new ArrayDeque<>();
        this.stateFactory = stateFactory;
        this.validator = new StateTransitionValidator();
    }
    
    /**
     * 推入新状态
     */
    public void pushState(StateType type, Object... params) {
        T currentState = getCurrentState();
        
        // 暂停当前状态
        if (currentState != null) {
            currentState.onPause();
        }
        
        // 创建新状态
        T newState = stateFactory.createState(type, params);
        
        // 验证状态转换
        if (!validator.canTransition(currentState, newState)) {
            throw new IllegalStateException("Invalid state transition");
        }
        
        // 推入栈并激活
        stateStack.push(newState);
        newState.onEnter(currentState);
        newState.onActivate();
    }
    
    /**
     * 弹出当前状态
     */
    public T popState() {
        if (stateStack.isEmpty()) {
            throw new IllegalStateException("Cannot pop from empty state stack");
        }
        
        T currentState = stateStack.pop();
        T previousState = getCurrentState();
        
        // 退出当前状态
        currentState.onDeactivate();
        currentState.onExit(previousState);
        
        // 恢复前一个状态
        if (previousState != null) {
            previousState.onResume(currentState);
        }
        
        return currentState;
    }
    
    /**
     * 替换当前状态
     */
    public void replaceState(StateType type, Object... params) {
        if (!stateStack.isEmpty()) {
            T currentState = stateStack.pop();
            currentState.onDeactivate();
            currentState.onExit(null);
        }
        
        pushState(type, params);
    }
    
    /**
     * 清空到指定状态
     */
    public void popToState(Class<? extends T> targetStateClass) {
        while (!stateStack.isEmpty()) {
            T currentState = getCurrentState();
            if (targetStateClass.isInstance(currentState)) {
                break;
            }
            popState();
        }
    }
    
    /**
     * 更新状态栈
     */
    public void update(float deltaTime) {
        T currentState = getCurrentState();
        if (currentState != null) {
            currentState.update(deltaTime);
        }
    }
    
    /**
     * 渲染状态栈
     */
    public void render() {
        // 从底到顶渲染所有可见状态
        List<T> visibleStates = new ArrayList<>();
        for (T state : stateStack) {
            visibleStates.add(state);
            if (!state.isTransparent()) {
                visibleStates.clear();
                visibleStates.add(state);
            }
        }
        
        for (T state : visibleStates) {
            state.render();
        }
    }
    
    public T getCurrentState() {
        return stateStack.isEmpty() ? null : stateStack.peek();
    }
    
    public int getStateCount() {
        return stateStack.size();
    }
    
    public boolean isEmpty() {
        return stateStack.isEmpty();
    }
}
```

### 2. 状态基类设计
```java
/**
 * 状态基类
 */
public abstract class State {
    
    protected StateStack<?> stateStack;
    protected StateType type;
    protected Map<String, Object> properties;
    protected boolean isActive;
    protected boolean isTransparent;
    
    public State(StateType type) {
        this.type = type;
        this.properties = new HashMap<>();
        this.isActive = false;
        this.isTransparent = false;
    }
    
    /**
     * 状态进入时调用
     */
    public void onEnter(State previousState) {
        System.out.println("Entering state: " + type);
    }
    
    /**
     * 状态激活时调用
     */
    public void onActivate() {
        isActive = true;
        System.out.println("Activating state: " + type);
    }
    
    /**
     * 状态暂停时调用
     */
    public void onPause() {
        isActive = false;
        System.out.println("Pausing state: " + type);
    }
    
    /**
     * 状态恢复时调用
     */
    public void onResume(State pausingState) {
        isActive = true;
        System.out.println("Resuming state: " + type);
    }
    
    /**
     * 状态停用时调用
     */
    public void onDeactivate() {
        isActive = false;
        System.out.println("Deactivating state: " + type);
    }
    
    /**
     * 状态退出时调用
     */
    public void onExit(State nextState) {
        System.out.println("Exiting state: " + type);
    }
    
    /**
     * 状态更新逻辑
     */
    public abstract void update(float deltaTime);
    
    /**
     * 状态渲染逻辑
     */
    public abstract void render();
    
    /**
     * 处理输入事件
     */
    public abstract boolean handleInput(InputEvent event);
    
    // Getter和Setter方法
    public StateType getType() { return type; }
    public boolean isActive() { return isActive; }
    public boolean isTransparent() { return isTransparent; }
    public void setTransparent(boolean transparent) { this.isTransparent = transparent; }
    
    public void setProperty(String key, Object value) {
        properties.put(key, value);
    }
    
    public <T> T getProperty(String key, Class<T> type) {
        Object value = properties.get(key);
        return type.isInstance(value) ? type.cast(value) : null;
    }
}
```

```java
 * 主菜单状态

public class MainMenuState extends State {
    
    private MenuUI menuUI;
    
    public MainMenuState() {
        super(GameStateType.MAIN_MENU);
        this.menuUI = new MenuUI();
    }
    
    @Override
    public void onEnter(State previousState) {
        super.onEnter(previousState);
        menuUI.initialize();
        // 播放背景音乐
        AudioManager.playBackgroundMusic("menu_theme.mp3");
    }
    
    @Override
    public void update(float deltaTime) {
        menuUI.update(deltaTime);
    }
    
    @Override
    public void render() {
        menuUI.render();
    }
    
    @Override
    public boolean handleInput(InputEvent event) {
        if (event.getType() == InputType.KEY_PRESSED) {
            switch (event.getKeyCode()) {
                case KeyEvent.VK_ENTER:
                    // 开始游戏
                    stateStack.pushState(GameStateType.LOADING, "game_level_1");
                    return true;
                    
                case KeyEvent.VK_S:
                    // 设置菜单
                    stateStack.pushState(GameStateType.SETTINGS);
                    return true;
                    
                case KeyEvent.VK_ESCAPE:
                    // 退出游戏
                    System.exit(0);
                    return true;
            }
        }
        return false;
    }
    
    @Override
    public void onExit(State nextState) {
        super.onExit(nextState);
        AudioManager.stopBackgroundMusic();
    }
}

/**

public class GamePlayingState extends State {
    
    private GameWorld gameWorld;
    private Player player;
    private Camera camera;
    
    public GamePlayingState(String levelName) {
        super(GameStateType.GAME_PLAYING);
        this.gameWorld = new GameWorld(levelName);
        this.player = new Player();
        this.camera = new Camera();
    }
    
    @Override
    public void onEnter(State previousState) {
        super.onEnter(previousState);
        gameWorld.initialize();
        player.spawn(gameWorld.getSpawnPoint());
        camera.followTarget(player);
        
        // 播放游戏音乐
        AudioManager.playBackgroundMusic("game_theme.mp3");
    }
    
    @Override
    public void update(float deltaTime) {
        if (!isActive) return;
        
        player.update(deltaTime);
        gameWorld.update(deltaTime);
        camera.update(deltaTime);
        
        // 检查游戏结束条件
        if (player.isDead()) {
            stateStack.pushState(GameStateType.GAME_OVER);
        }
    }
    
    @Override
    public void render() {
        camera.begin();
        gameWorld.render();
        player.render();
        camera.end();
        
        // 渲染UI
        renderGameUI();
    }
    
    @Override
    public boolean handleInput(InputEvent event) {
        if (event.getType() == InputType.KEY_PRESSED) {
            switch (event.getKeyCode()) {
                case KeyEvent.VK_ESCAPE:
                    // 暂停游戏
                    stateStack.pushState(GameStateType.PAUSE_MENU);
                    return true;
                    
                case KeyEvent.VK_I:
                    // 打开物品栏
                    stateStack.pushState(GameStateType.INVENTORY);
                    return true;
            }
        }
        
        // 传递给玩家处理
        return player.handleInput(event);
    }
    
    private void renderGameUI() {
        // 渲染血条、分数等UI元素
        UIRenderer.renderHealthBar(player.getHealth(), player.getMaxHealth());
        UIRenderer.renderScore(player.getScore());
    }
    
    @Override
    public void onPause() {
        super.onPause();
        gameWorld.pause();
        AudioManager.pauseBackgroundMusic();
    }
    
    @Override
    public void onResume(State pausingState) {
        super.onResume(pausingState);
        gameWorld.resume();
        AudioManager.resumeBackgroundMusic();
    }
}

/**
```
```java
 * 游戏状态工厂
 */
public class GameStateFactory implements StateFactory<State> {
    
    @Override
    public State createState(StateType type, Object... params) {
        if (!(type instanceof GameStateType)) {
            throw new IllegalArgumentException("Invalid state type");
        }
        
        GameStateType gameStateType = (GameStateType) type;
        
        switch (gameStateType) {
            case MAIN_MENU:
                return new MainMenuState();
                
            case GAME_PLAYING:
                String levelName = params.length > 0 ? (String) params[0] : "default";
                return new GamePlayingState(levelName);
                
            case PAUSE_MENU:
                return new PauseMenuState();
                
            case SETTINGS:
                return new SettingsState();
                
            case GAME_OVER:
                int score = params.length > 0 ? (Integer) params[0] : 0;
                return new GameOverState(score);
                
            case LOADING:
                String targetLevel = params.length > 0 ? (String) params[0] : "";
                return new LoadingState(targetLevel);
                
            default:
                throw new IllegalArgumentException("Unknown state type: " + type);
        }
    }
}
```

```java
/**
 * 状态机状态栈
 * 结合有限状态机和状态栈的优势
 */
public class StateMachineStack<T extends State> extends StateStack<T> {
    
    private final Map<StateType, Set<StateType>> allowedTransitions;
    private final Map<StateType, StateTransitionHandler> transitionHandlers;
    
    public StateMachineStack(StateFactory<T> stateFactory) {
        super(stateFactory);
        this.allowedTransitions = new HashMap<>();
        this.transitionHandlers = new HashMap<>();
        initializeTransitions();
    }
    
    private void initializeTransitions() {
        // 定义允许的状态转换
        addTransition(GameStateType.MAIN_MENU, 
                     Set.of(GameStateType.LOADING, GameStateType.SETTINGS));
        
        addTransition(GameStateType.LOADING, 
                     Set.of(GameStateType.GAME_PLAYING, GameStateType.MAIN_MENU));
        
        addTransition(GameStateType.GAME_PLAYING, 
                     Set.of(GameStateType.PAUSE_MENU, GameStateType.GAME_OVER));
        
        addTransition(GameStateType.PAUSE_MENU, 
                     Set.of(GameStateType.GAME_PLAYING, GameStateType.SETTINGS, 
                            GameStateType.MAIN_MENU));
        
        // 设置转换处理器
        setTransitionHandler(GameStateType.LOADING, new LoadingTransitionHandler());
        setTransitionHandler(GameStateType.GAME_OVER, new GameOverTransitionHandler());
    }
    
    private void addTransition(StateType from, Set<StateType> toStates) {
        allowedTransitions.put(from, toStates);
    }
    
    private void setTransitionHandler(StateType stateType, StateTransitionHandler handler) {
        transitionHandlers.put(stateType, handler);
    }
    
    @Override
    public void pushState(StateType type, Object... params) {
        T currentState = getCurrentState();
        
        // 检查状态转换是否被允许
        if (currentState != null) {
            Set<StateType> allowed = allowedTransitions.get(currentState.getType());
            if (allowed == null || !allowed.contains(type)) {
                throw new IllegalStateException(
                    String.format("Transition from %s to %s is not allowed", 
                                currentState.getType(), type));
            }
        }
        
        // 执行转换前处理
        StateTransitionHandler handler = transitionHandlers.get(type);
        if (handler != null) {
            handler.beforeTransition(currentState, type, params);
        }
        
        super.pushState(type, params);
        
        // 执行转换后处理
        if (handler != null) {
            handler.afterTransition(getCurrentState());
        }
    }
    
    /**
     * 状态转换处理器接口
     */
    public interface StateTransitionHandler {
        void beforeTransition(State currentState, StateType targetType, Object... params);
        void afterTransition(State newState);
    }
    
    /**
     * 加载状态转换处理器
     */
    private static class LoadingTransitionHandler implements StateTransitionHandler {
        
        @Override
        public void beforeTransition(State currentState, StateType targetType, Object... params) {
            // 保存当前游戏状态
            if (currentState instanceof GamePlayingState) {
                GameSaveManager.saveGame((GamePlayingState) currentState);
            }
        }
        
        @Override
        public void afterTransition(State newState) {
            if (newState instanceof LoadingState) {
                LoadingState loadingState = (LoadingState) newState;
                loadingState.startLoading();
            }
        }
    }
    
    /**
     * 游戏结束状态转换处理器
     */
    private static class GameOverTransitionHandler implements StateTransitionHandler {
        
        @Override
        public void beforeTransition(State currentState, StateType targetType, Object... params) {
            // 记录游戏统计信息
            if (currentState instanceof GamePlayingState) {
                GamePlayingState gameState = (GamePlayingState) currentState;
                GameStatistics.recordGameEnd(gameState.getPlayer().getScore(), 
                                           gameState.getPlayTime());
            }
        }
        
        @Override
        public void afterTransition(State newState) {
            // 显示游戏结束动画
            if (newState instanceof GameOverState) {
                ((GameOverState) newState).playGameOverAnimation();
            }
        }
    }
}
```

### 2. 状态持久化和恢复
```java
/**
 * 可持久化的状态栈
 */
public class PersistentStateStack<T extends State> extends StateStack<T> {
    
    private final StateSerializer<T> serializer;
    private final String saveFilePath;
    
    public PersistentStateStack(StateFactory<T> stateFactory, 
                               StateSerializer<T> serializer,
                               String saveFilePath) {
        super(stateFactory);
        this.serializer = serializer;
        this.saveFilePath = saveFilePath;
    }
    
    /**
     * 保存状态栈到文件
     */
    public void saveToFile() throws IOException {
        List<StateSnapshot> snapshots = new ArrayList<>();
        
        for (T state : stateStack) {
            StateSnapshot snapshot = serializer.serialize(state);
            snapshots.add(snapshot);
        }
        
        try (ObjectOutputStream oos = new ObjectOutputStream(
                new FileOutputStream(saveFilePath))) {
            oos.writeObject(snapshots);
        }
    }
    
    /**
     * 从文件恢复状态栈
     */
    @SuppressWarnings("unchecked")
    public void loadFromFile() throws IOException, ClassNotFoundException {
        try (ObjectInputStream ois = new ObjectInputStream(
                new FileInputStream(saveFilePath))) {
            
            List<StateSnapshot> snapshots = (List<StateSnapshot>) ois.readObject();
            
            // 清空当前状态栈
            while (!isEmpty()) {
                popState();
            }
            
            // 重建状态栈
            for (StateSnapshot snapshot : snapshots) {
                T state = serializer.deserialize(snapshot);
                stateStack.addLast(state); // 从底部添加
            }
            
            // 激活顶部状态
            T topState = getCurrentState();
            if (topState != null) {
                topState.onActivate();
            }
        }
    }
    
    /**
     * 创建状态快照
     */
    public StateStackSnapshot createSnapshot() {
        List<StateSnapshot> stateSnapshots = new ArrayList<>();
        
        for (T state : stateStack) {
            stateSnapshots.add(serializer.serialize(state));
        }
        
        return new StateStackSnapshot(stateSnapshots);
    }
    
    /**
     * 恢复到指定快照
     */
    public void restoreFromSnapshot(StateStackSnapshot snapshot) {
        // 清空当前状态栈
        while (!isEmpty()) {
            popState();
        }
        
        // 重建状态栈
        for (StateSnapshot stateSnapshot : snapshot.getStateSnapshots()) {
            T state = serializer.deserialize(stateSnapshot);
            stateStack.addLast(state);
        }
        
        // 激活顶部状态
        T topState = getCurrentState();
        if (topState != null) {
            topState.onActivate();
        }
    }
    
    /**
     * 状态序列化器接口
     */
    public interface StateSerializer<T extends State> {
        StateSnapshot serialize(T state);
        T deserialize(StateSnapshot snapshot);
    }
    
    /**
     * 状态快照
     */
    public static class StateSnapshot implements Serializable {
        private final StateType type;
        private final Map<String, Object> properties;
        private final byte[] stateData;
        
        public StateSnapshot(StateType type, Map<String, Object> properties, byte[] stateData) {
            this.type = type;
            this.properties = new HashMap<>(properties);
            this.stateData = stateData.clone();
        }
        
        // Getter方法
        public StateType getType() { return type; }
        public Map<String, Object> getProperties() { return new HashMap<>(properties); }
        public byte[] getStateData() { return stateData.clone(); }
    }
    
    /**
     * 状态栈快照
     */
    public static class StateStackSnapshot {
        private final List<StateSnapshot> stateSnapshots;
        private final long timestamp;
        
        public StateStackSnapshot(List<StateSnapshot> stateSnapshots) {
            this.stateSnapshots = new ArrayList<>(stateSnapshots);
            this.timestamp = System.currentTimeMillis();
        }
        
        public List<StateSnapshot> getStateSnapshots() { 
            return new ArrayList<>(stateSnapshots); 
        }
        
        public long getTimestamp() { return timestamp; }
    }
}
```

### 3. 异步状态加载
```java
/**
 * 支持异步加载的状态栈
 */
public class AsyncStateStack<T extends State> extends StateStack<T> {
    
    private final ExecutorService loadingExecutor;
    private final Map<StateType, CompletableFuture<T>> loadingStates;
    
    public AsyncStateStack(StateFactory<T> stateFactory) {
        super(stateFactory);
        this.loadingExecutor = Executors.newCachedThreadPool();
        this.loadingStates = new ConcurrentHashMap<>();
    }
    
    /**
     * 异步预加载状态
     */
    public CompletableFuture<T> preloadState(StateType type, Object... params) {
        return loadingStates.computeIfAbsent(type, k -> 
            CompletableFuture.supplyAsync(() -> {
                try {
                    T state = stateFactory.createState(type, params);
                    state.preload(); // 预加载资源
                    return state;
                } catch (Exception e) {
                    throw new RuntimeException("Failed to preload state: " + type, e);
                }
            }, loadingExecutor)
        );
    }
    
    /**
     * 异步推入状态
     */
    public CompletableFuture<Void> pushStateAsync(StateType type, Object... params) {
        CompletableFuture<T> stateFuture = loadingStates.get(type);
        
        if (stateFuture == null) {
            // 如果没有预加载，立即开始加载
            stateFuture = preloadState(type, params);
        }
        
        return stateFuture.thenAccept(state -> {
            // 在主线程中推入状态
            Platform.runLater(() -> {
                loadingStates.remove(type);
                pushPreloadedState(state);
            });
        });
    }
    
    private void pushPreloadedState(T state) {
        T currentState = getCurrentState();
        
        if (currentState != null) {
            currentState.onPause();
        }
        
        stateStack.push(state);
        state.onEnter(currentState);
        state.onActivate();
    }
    
    /**
     * 批量预加载状态
     */
    public CompletableFuture<Void> preloadStates(StateType... types) {
        CompletableFuture<?>[] futures = Arrays.stream(types)
            .map(this::preloadState)
            .toArray(CompletableFuture[]::new);
        
        return CompletableFuture.allOf(futures);
    }
    
    /**
     * 获取加载进度
     */
    public LoadingProgress getLoadingProgress() {
        int total = loadingStates.size();
        int completed = 0;
        
        for (CompletableFuture<T> future : loadingStates.values()) {
            if (future.isDone()) {
                completed++;
            }
        }
        
        return new LoadingProgress(completed, total);
    }
    
    public void shutdown() {
        loadingExecutor.shutdown();
        try {
            if (!loadingExecutor.awaitTermination(5, TimeUnit.SECONDS)) {
                loadingExecutor.shutdownNow();
            }
        } catch (InterruptedException e) {
            loadingExecutor.shutdownNow();
            Thread.currentThread().interrupt();
        }
    }
    
    public static class LoadingProgress {
        private final int completed;
        private final int total;
        
        public LoadingProgress(int completed, int total) {
            this.completed = completed;
            this.total = total;
        }
        
        public int getCompleted() { return completed; }
        public int getTotal() { return total; }
        public float getPercentage() { 
            return total == 0 ? 1.0f : (float) completed / total; 
        }
        public boolean isComplete() { return completed == total; }
    }
}
```

| **状态恢复** | 不支持 | 自动恢复 |
| **内存使用** | 较少 | 较多 |
| **复杂度** | 简单 | 复杂 |


### 2. 命令模式集成
```java
/**
 * 支持命令模式的状态栈
 */
public class CommandableStateStack<T extends State> extends StateStack<T> {
    
    private final Deque<StateCommand> commandHistory;
    private final int maxHistorySize;
    
    public CommandableStateStack(StateFactory<T> stateFactory, int maxHistorySize) {
        super(stateFactory);
        this.commandHistory = new ArrayDeque<>();
        this.maxHistorySize = maxHistorySize;
    }
    
    /**
     * 执行状态命令
     */
    public void executeCommand(StateCommand command) {
        command.execute(this);
        
        // 添加到历史记录
        commandHistory.push(command);
        
        // 限制历史记录大小
        while (commandHistory.size() > maxHistorySize) {
            commandHistory.removeLast();
        }
    }
    
    /**
     * 撤销上一个命令
     */
    public boolean undo() {
        if (commandHistory.isEmpty()) {
            return false;
        }
        
        StateCommand lastCommand = commandHistory.pop();
        lastCommand.undo(this);
        return true;
    }
    
    /**
     * 状态命令接口
     */
    public interface StateCommand {
        void execute(StateStack<?> stateStack);
        void undo(StateStack<?> stateStack);
        String getDescription();
    }
    
    /**
     * 推入状态命令
     */
    public static class PushStateCommand implements StateCommand {
        private final StateType stateType;
        private final Object[] params;
        private State pushedState;
        
        public PushStateCommand(StateType stateType, Object... params) {
            this.stateType = stateType;
            this.params = params;
        }
        
        @Override
        public void execute(StateStack<?> stateStack) {
            stateStack.pushState(stateType, params);
            pushedState = stateStack.getCurrentState();
        }
        
        @Override
        public void undo(StateStack<?> stateStack) {
            if (pushedState != null && stateStack.getCurrentState() == pushedState) {
                stateStack.popState();
            }
        }
        
        @Override
        public String getDescription() {
            return "Push state: " + stateType;
        }
    }
    
    /**
     * 弹出状态命令
     */
    public static class PopStateCommand implements StateCommand {
        private State poppedState;
        private StateType poppedStateType;
        
        @Override
        public void execute(StateStack<?> stateStack) {
            poppedState = stateStack.getCurrentState();
            if (poppedState != null) {
                poppedStateType = poppedState.getType();
                stateStack.popState();
            }
        }
        
        @Override
        public void undo(StateStack<?> stateStack) {
            if (poppedStateType != null) {
                stateStack.pushState(poppedStateType);
            }
        }
        
        @Override
        public String getDescription() {
            return "Pop state: " + (poppedStateType != null ? poppedStateType : "unknown");
        }
    }
}
```

```java
/**
 * 高性能状态栈实现
 */
public class OptimizedStateStack<T extends State> extends StateStack<T> {
    
    // 状态对象池
    private final Map<StateType, Queue<T>> statePool;
    
    // 状态缓存
    private final LRUCache<String, T> stateCache;
    
    // 性能监控
    private final StateStackProfiler profiler;
    
    public OptimizedStateStack(StateFactory<T> stateFactory, int maxCacheSize) {
        super(stateFactory);
        this.statePool = new ConcurrentHashMap<>();
        this.stateCache = new LRUCache<>(maxCacheSize);
        this.profiler = new StateStackProfiler();
    }
    
    @Override
    public void pushState(StateType type, Object... params) {
        long startTime = System.nanoTime();
        
        try {
            T state = acquireState(type, params);
            super.pushPreloadedState(state);
        } finally {
            profiler.recordPushTime(System.nanoTime() - startTime);
        }
    }
    
    @Override
    public T popState() {
        long startTime = System.nanoTime();
        
        try {
            T state = super.popState();
            releaseState(state);
            return state;
        } finally {
            profiler.recordPopTime(System.nanoTime() - startTime);
        }
    }
    
    private T acquireState(StateType type, Object... params) {
        // 尝试从对象池获取
        Queue<T> pool = statePool.get(type);
        if (pool != null && !pool.isEmpty()) {
            T state = pool.poll();
            state.reset(params);
            return state;
        }
        
        // 尝试从缓存获取
        String cacheKey = createCacheKey(type, params);
        T cachedState = stateCache.get(cacheKey);
        if (cachedState != null) {
            return cachedState;
        }
        
        // 创建新状态
        T newState = stateFactory.createState(type, params);
        stateCache.put(cacheKey, newState);
        return newState;
    }
    
    private void releaseState(T state) {
        if (state.isReusable()) {
            Queue<T> pool = statePool.computeIfAbsent(
                state.getType(), k -> new ConcurrentLinkedQueue<>());
            
            state.cleanup();
            pool.offer(state);
        }
    }
    
    private String createCacheKey(StateType type, Object... params) {
        StringBuilder key = new StringBuilder(type.toString());
        for (Object param : params) {
            key.append("_").append(param.toString());
        }
        return key.toString();
    }
    
    /**
     * 性能分析器
     */
    private static class StateStackProfiler {
        private final AtomicLong totalPushTime = new AtomicLong();
        private final AtomicLong totalPopTime = new AtomicLong();
        private final AtomicLong pushCount = new AtomicLong();
        private final AtomicLong popCount = new AtomicLong();
        
        void recordPushTime(long nanos) {
            totalPushTime.addAndGet(nanos);
            pushCount.incrementAndGet();
        }
        
        void recordPopTime(long nanos) {
            totalPopTime.addAndGet(nanos);
            popCount.incrementAndGet();
        }
        
        public double getAveragePushTime() {
            long count = pushCount.get();
            return count == 0 ? 0 : (double) totalPushTime.get() / count / 1_000_000; // ms
        }
        
        public double getAveragePopTime() {
            long count = popCount.get();
            return count == 0 ? 0 : (double) totalPopTime.get() / count / 1_000_000; // ms
        }
    }
}
```
