```java
    public boolean isValidParentheses(String s) {
        Deque<Character> stack = new ArrayDeque<>();
        Map<Character, Character> pairs = Map.of(')', '(', ']', '[', '}', '{');
        
        for (char c : s.toCharArray()) {
            if (pairs.containsValue(c)) {
                stack.push(c);
            } else if (pairs.containsKey(c)) {
                if (stack.isEmpty() || stack.pop() != pairs.get(c)) {
                    return false;
                }
            }
        }
        
        return stack.isEmpty();
    }
}
```

```java
    public int[] nextGreaterElement(int[] nums) {
        int n = nums.length;
        int[] result = new int[n];
        Deque<Integer> stack = new ArrayDeque<>();
        
        for (int i = n - 1; i >= 0; i--) {
            while (!stack.isEmpty() && stack.peek() <= nums[i]) {
```

```java
public class AdvancedStackPractice {
    
    // 练习1：字符串解码
    public String decodeString(String s) {
        Deque<Integer> countStack = new ArrayDeque<>();
        Deque<StringBuilder> stringStack = new ArrayDeque<>();
        StringBuilder currentString = new StringBuilder();
        int count = 0;
        
        for (char c : s.toCharArray()) {
            if (Character.isDigit(c)) {
                count = count * 10 + (c - '0');
            } else if (c == '[') {
                countStack.push(count);
                stringStack.push(currentString);
                count = 0;
                currentString = new StringBuilder();
            } else if (c == ']') {
                StringBuilder temp = currentString;
                currentString = stringStack.pop();
                int repeatCount = countStack.pop();
                for (int i = 0; i < repeatCount; i++) {
                    currentString.append(temp);
                }
            } else {
                currentString.append(c);
            }
        }
        
        return currentString.toString();
    }
    
    // 练习2：接雨水
    public int trap(int[] height) {
        Deque<Integer> stack = new ArrayDeque<>();
        int water = 0;
        
        for (int i = 0; i < height.length; i++) {
            while (!stack.isEmpty() && height[i] > height[stack.peek()]) {
                int bottom = stack.pop();
                if (stack.isEmpty()) break;
                
                int distance = i - stack.peek() - 1;
                int boundedHeight = Math.min(height[i], height[stack.peek()]) - height[bottom];
                water += distance * boundedHeight;
            }
            stack.push(i);
        }
        
        return water;
    }
}
```

```java
class MinStack {
    private Deque<Integer> dataStack;
    private Deque<Integer> minStack;
    
    public MinStack() {
        dataStack = new ArrayDeque<>();
        minStack = new ArrayDeque<>();
    }
    
    public void push(int val) {
        dataStack.push(val);
        if (minStack.isEmpty() || val <= minStack.peek()) {
            minStack.push(val);
        }
    }
    
    public void pop() {
        int val = dataStack.pop();
        if (val == minStack.peek()) {
            minStack.pop();
        }
    }
    
    public int top() {
        return dataStack.peek();
    }
    
    public int getMin() {
        return minStack.peek();
    }
}
```

### 项目1：表达式计算器
**项目描述：** 实现一个支持四则运算、括号、函数调用的表达式计算器

**技术要点：**

+ 中缀表达式转后缀表达式
+ 后缀表达式求值
+ 函数调用处理
+ 错误处理机制

**实现代码：**

```java
public class ExpressionCalculator {
    private Map<String, Function<Double[], Double>> functions;
    
    public ExpressionCalculator() {
        initializeFunctions();
    }
    
    public double evaluate(String expression) {
        String postfix = infixToPostfix(expression);
        return evaluatePostfix(postfix);
    }
    
    private String infixToPostfix(String infix) {
        // 实现中缀转后缀算法
        StringBuilder postfix = new StringBuilder();
        Deque<String> stack = new ArrayDeque<>();
        
        String[] tokens = tokenize(infix);
        for (String token : tokens) {
            if (isNumber(token)) {
                postfix.append(token).append(" ");
            } else if (isFunction(token)) {
                stack.push(token);
            } else if (token.equals("(")) {
                stack.push(token);
            } else if (token.equals(")")) {
                while (!stack.isEmpty() && !stack.peek().equals("(")) {
                    postfix.append(stack.pop()).append(" ");
                }
                stack.pop(); // 移除 "("
                if (!stack.isEmpty() && isFunction(stack.peek())) {
                    postfix.append(stack.pop()).append(" ");
                }
            } else if (isOperator(token)) {
                while (!stack.isEmpty() && !stack.peek().equals("(") &&
                       getPrecedence(stack.peek()) >= getPrecedence(token)) {
                    postfix.append(stack.pop()).append(" ");
                }
                stack.push(token);
            }
        }
        
        while (!stack.isEmpty()) {
            postfix.append(stack.pop()).append(" ");
        }
        
        return postfix.toString().trim();
    }
    
    private double evaluatePostfix(String postfix) {
        // 实现后缀表达式求值
        Deque<Double> stack = new ArrayDeque<>();
        String[] tokens = postfix.split(" ");
        
        for (String token : tokens) {
            if (isNumber(token)) {
                stack.push(Double.parseDouble(token));
            } else if (isOperator(token)) {
                double b = stack.pop();
                double a = stack.pop();
                stack.push(applyOperator(token, a, b));
            } else if (isFunction(token)) {
                int paramCount = getFunctionParamCount(token);
                Double[] params = new Double[paramCount];
                for (int i = paramCount - 1; i >= 0; i--) {
                    params[i] = stack.pop();
                }
                stack.push(functions.get(token).apply(params));
            }
        }
        
        return stack.pop();
    }
    
    private void initializeFunctions() {
        functions = new HashMap<>();
        functions.put("sin", params -> Math.sin(params[0]));
```

### 项目2：文本编辑器撤销重做
**项目描述：** 实现一个支持撤销重做功能的文本编辑器

**技术要点：**

+ 命令模式设计
+ 双栈实现撤销重做
+ 内存优化策略
+ 批量操作处理

**实现代码：**

```java
public class TextEditor {
    private StringBuilder content;
    private Deque<Command> undoStack;
    private Deque<Command> redoStack;
    private int maxUndoSteps;
    
    public TextEditor(int maxUndoSteps) {
        this.content = new StringBuilder();
        this.undoStack = new ArrayDeque<>();
        this.redoStack = new ArrayDeque<>();
        this.maxUndoSteps = maxUndoSteps;
    }
    
    public void insert(int position, String text) {
        executeCommand(new InsertCommand(position, text));
    }
    
    public void delete(int position, int length) {
        executeCommand(new DeleteCommand(position, length));
    }
    
    public void replace(int position, int length, String newText) {
        executeCommand(new ReplaceCommand(position, length, newText));
    }
    
    public void undo() {
        if (!undoStack.isEmpty()) {
            Command command = undoStack.pop();
            command.undo(content);
            redoStack.push(command);
        }
    }
    
    public void redo() {
        if (!redoStack.isEmpty()) {
            Command command = redoStack.pop();
            command.execute(content);
            undoStack.push(command);
        }
    }
    
    private void executeCommand(Command command) {
        command.execute(content);
        undoStack.push(command);
        redoStack.clear();
        
        // 限制撤销步数
        if (undoStack.size() > maxUndoSteps) {
            // 移除最早的命令
            Deque<Command> temp = new ArrayDeque<>();
            for (int i = 0; i < maxUndoSteps; i++) {
                temp.push(undoStack.pop());
            }
            undoStack.clear();
            while (!temp.isEmpty()) {
                undoStack.addLast(temp.pop());
            }
        }
    }
    
    // 命令接口和实现类...
    interface Command {
        void execute(StringBuilder content);
        void undo(StringBuilder content);
    }
    
    class InsertCommand implements Command {
        private int position;
        private String text;
        
        public InsertCommand(int position, String text) {
            this.position = position;
            this.text = text;
        }
        
        @Override
        public void execute(StringBuilder content) {
            content.insert(position, text);
        }
        
        @Override
        public void undo(StringBuilder content) {
            content.delete(position, position + text.length());
        }
    }
```

### 项目3：浏览器历史记录管理
**项目描述：** 实现一个浏览器历史记录管理系统

**技术要点：**

+ 双栈管理前进后退
+ URL验证和处理
+ 历史记录持久化
+ 内存管理优化

**实现代码：**

```java
public class BrowserHistory {
    private Deque<String> backStack;
    private Deque<String> forwardStack;
    private String currentUrl;
    private int maxHistorySize;
    
    public BrowserHistory(String homepage, int maxHistorySize) {
        this.backStack = new ArrayDeque<>();
        this.forwardStack = new ArrayDeque<>();
        this.currentUrl = homepage;
        this.maxHistorySize = maxHistorySize;
    }
    
    public void visit(String url) {
        if (currentUrl != null) {
            backStack.push(currentUrl);
            limitStackSize(backStack);
        }
        currentUrl = url;
        forwardStack.clear();
    }
    
    public String back(int steps) {
        for (int i = 0; i < steps && !backStack.isEmpty(); i++) {
            forwardStack.push(currentUrl);
            currentUrl = backStack.pop();
        }
        limitStackSize(forwardStack);
        return currentUrl;
    }
    
    public String forward(int steps) {
        for (int i = 0; i < steps && !forwardStack.isEmpty(); i++) {
            backStack.push(currentUrl);
            currentUrl = forwardStack.pop();
        }
        limitStackSize(backStack);
        return currentUrl;
    }
    
    public String getCurrentUrl() {
        return currentUrl;
    }
    
    public List<String> getHistory() {
        List<String> history = new ArrayList<>();
        
        // 添加后退历史
        List<String> backHistory = new ArrayList<>(backStack);
        Collections.reverse(backHistory);
        history.addAll(backHistory);
        
        // 添加当前页面
        history.add(currentUrl);
        
        // 添加前进历史
        history.addAll(forwardStack);
        
        return history;
    }
    
    private void limitStackSize(Deque<String> stack) {
        while (stack.size() > maxHistorySize) {
            // 移除最早的记录
            String[] temp = stack.toArray(new String[0]);
            stack.clear();
            for (int i = 1; i < temp.length; i++) {
                stack.addLast(temp[i]);
            }
        }
    }
    
    // 历史记录搜索
    public List<String> searchHistory(String keyword) {
        return getHistory().stream()
                .filter(url -> url.toLowerCase().contains(keyword.toLowerCase()))
                .collect(Collectors.toList());
    }
    
    // 清除历史记录
    public void clearHistory() {
        backStack.clear();
        forwardStack.clear();
    }
}
```

## ❓ 常见问题解答
### Q1: 什么时候应该使用栈？
**A:** 栈适用于以下场景：

+ 需要后进先出的数据处理
+ 括号匹配、表达式求值
+ 递归算法的迭代实现
+ 函数调用管理
+ 撤销重做功能
+ 深度优先搜索

### Q2: 如何选择栈的实现方式？
**A:** 选择依据：

+ **ArrayDeque**：推荐，性能最好，适合大多数场景
+ **LinkedList**：内存动态分配，适合大小变化频繁的场景
+ **Stack类**：不推荐，继承Vector，性能较差
+ **数组模拟**：适合基本类型，避免装箱拆箱

### Q3: 单调栈什么时候使用？
**A:** 单调栈适用于：

+ 寻找下一个更大/更小元素
+ 柱状图最大矩形问题
+ 接雨水问题
+ 需要维护序列单调性的场景

### Q4: 如何调试栈算法？
**A:** 调试技巧：

+ 打印栈的状态变化
+ 使用可视化工具
+ 手动模拟算法过程
+ 添加断言检查栈的不变量
+ 使用单元测试验证边界情况

### Q5: 栈溢出如何避免？
**A:** 预防措施：

+ 限制递归深度
+ 使用迭代替代递归
+ 增加栈空间大小
+ 优化算法减少栈使用
+ 添加栈大小检查

### Q6: 如何优化栈算法的性能？
**A:** 优化策略：

+ 选择合适的数据结构
+ 避免不必要的对象创建
+ 使用基本类型数组模拟栈
+ 提前终止条件判断
+ 复用对象减少GC压力

### Q7: 栈在实际项目中有哪些应用？
**A:** 实际应用：

+ 编译器的语法分析
+ 浏览器的前进后退功能
+ 文本编辑器的撤销重做
+ 函数调用栈管理
+ 表达式求值引擎
+ 深度优先搜索算法
