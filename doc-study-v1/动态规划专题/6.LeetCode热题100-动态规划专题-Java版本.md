```java
    dp[1] = 1;
    dp[2] = 2;

public int climbStairsOptimized(int n) {

        prev2 = prev1;

    }

}
```

```java
public int maxSubArray(int[] nums) {
    int n = nums.length;
    int[] dp = new int[n];
    dp[0] = nums[0];
    int maxSum = dp[0];
    
    for (int i = 1; i < n; i++) {
        dp[i] = Math.max(nums[i], dp[i - 1] + nums[i]);
        maxSum = Math.max(maxSum, dp[i]);
    }
    return maxSum;
}

}
```

```java
public int maxProduct(int[] nums) {
    int n = nums.length;
    int[] maxDp = new int[n];
    int[] minDp = new int[n];
    
    maxDp[0] = minDp[0] = nums[0];
    int result = nums[0];
    
    for (int i = 1; i < n; i++) {
        maxDp[i] = Math.max(nums[i], Math.max(maxDp[i-1] * nums[i], minDp[i-1] * nums[i]));
        minDp[i] = Math.min(nums[i], Math.min(maxDp[i-1] * nums[i], minDp[i-1] * nums[i]));
        result = Math.max(result, maxDp[i]);
    }
    return result;
}

// 空间优化版本
public int maxProductOptimized(int[] nums) {
    int maxProd = nums[0], minProd = nums[0], result = nums[0];
    
    for (int i = 1; i < nums.length; i++) {
        int tempMax = Math.max(nums[i], Math.max(maxProd * nums[i], minProd * nums[i]));
        minProd = Math.min(nums[i], Math.min(maxProd * nums[i], minProd * nums[i]));
        maxProd = tempMax;
        result = Math.max(result, maxProd);
    }
    return result;
}
```

    - **Java模板**:
    - **难点**: 负数会使最大变最小，最小变最大

```java
public int rob(int[] nums) {
    int n = nums.length;
    if (n == 1) return nums[0];
    
    int[] dp = new int[n];
    dp[0] = nums[0];
    dp[1] = Math.max(nums[0], nums[1]);
    
    for (int i = 2; i < n; i++) {
        dp[i] = Math.max(dp[i - 1], dp[i - 2] + nums[i]);
    }
    return dp[n - 1];
}

// 空间优化版本
public int robOptimized(int[] nums) {
    int n = nums.length;
    if (n == 1) return nums[0];
    
    int prev2 = nums[0];
    int prev1 = Math.max(nums[0], nums[1]);
    
    for (int i = 2; i < n; i++) {
        int curr = Math.max(prev1, prev2 + nums[i]);
        prev2 = prev1;
        prev1 = curr;
    }
    return prev1;
}
```

```java
// O(n²) 解法
public int lengthOfLIS(int[] nums) {
    int n = nums.length;
    int[] dp = new int[n];
    Arrays.fill(dp, 1);
    
    int maxLen = 1;
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < i; j++) {
            if (nums[j] < nums[i]) {
                dp[i] = Math.max(dp[i], dp[j] + 1);
            }
        }
        maxLen = Math.max(maxLen, dp[i]);
    }
    return maxLen;
}

// O(nlogn) 二分优化解法
public int lengthOfLISOptimized(int[] nums) {
    List<Integer> tails = new ArrayList<>();
    
    for (int num : nums) {
        int left = 0, right = tails.size();
        while (left < right) {
            int mid = left + (right - left) / 2;
            if (tails.get(mid) < num) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        
        if (left == tails.size()) {
            tails.add(num);
        } else {
            tails.set(left, num);
        }
    }
    return tails.size();
}
```

    - **标签**: LIS、二分优化
    - **Java模板**:
    - **难点**: O(n²)到O(nlogn)的优化
    - **扩展**: 最长公共子序列、俄罗斯套娃

#### 💰 股票交易DP
+ ***� 309-最佳买卖股票时机含冷冻期**

```java
public int maxProfit(int[] prices) {
    int n = prices.length;
    if (n <= 1) return 0;
    
    // hold: 持有股票, sold: 卖出股票, rest: 冷冻期
    int[] hold = new int[n];
    int[] sold = new int[n];
    int[] rest = new int[n];
    
    hold[0] = -prices[0];
    sold[0] = 0;
    rest[0] = 0;
    
    for (int i = 1; i < n; i++) {
        hold[i] = Math.max(hold[i-1], rest[i-1] - prices[i]);
        sold[i] = hold[i-1] + prices[i];
        rest[i] = Math.max(rest[i-1], sold[i-1]);
    }
    
    return Math.max(sold[n-1], rest[n-1]);
}

// 空间优化版本
public int maxProfitOptimized(int[] prices) {
    int hold = -prices[0], sold = 0, rest = 0;
    
    for (int i = 1; i < prices.length; i++) {
        int prevHold = hold, prevSold = sold, prevRest = rest;
        hold = Math.max(prevHold, prevRest - prices[i]);
        sold = prevHold + prices[i];
        rest = Math.max(prevRest, prevSold);
    }
    
    return Math.max(sold, rest);
}
```

    - **标签**: 状态机DP、多状态转移
    - **Java模板**:
    - **难点**: 状态之间的转移关系
    - **扩展**: 多次交易、手续费

```java
public int numSquares(int n) {
    int[] dp = new int[n + 1];
    Arrays.fill(dp, Integer.MAX_VALUE);
    dp[0] = 0;
    
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j * j <= i; j++) {
            dp[i] = Math.min(dp[i], dp[i - j * j] + 1);
        }
    }
    return dp[n];
}
```

```java
public int coinChange(int[] coins, int amount) {
    int[] dp = new int[amount + 1];
    Arrays.fill(dp, amount + 1);
    dp[0] = 0;
    
    for (int i = 1; i <= amount; i++) {
        for (int coin : coins) {
            if (coin <= i) {
                dp[i] = Math.min(dp[i], dp[i - coin] + 1);
            }
        }
    }
    
    return dp[amount] > amount ? -1 : dp[amount];
}
```

```java
public boolean canPartition(int[] nums) {
    int sum = Arrays.stream(nums).sum();
    if (sum % 2 != 0) return false;
    
    int target = sum / 2;
    boolean[][] dp = new boolean[nums.length + 1][target + 1];
    
    // 初始化
    for (int i = 0; i <= nums.length; i++) {
        dp[i][0] = true;
    }
    
    for (int i = 1; i <= nums.length; i++) {
        for (int j = 1; j <= target; j++) {
            dp[i][j] = dp[i-1][j];
            if (j >= nums[i-1]) {
                dp[i][j] = dp[i][j] || dp[i-1][j-nums[i-1]];
            }
        }
    }
    
    return dp[nums.length][target];
}

// 空间优化版本
public boolean canPartitionOptimized(int[] nums) {
    int sum = Arrays.stream(nums).sum();
    if (sum % 2 != 0) return false;
    
    int target = sum / 2;
    boolean[] dp = new boolean[target + 1];
    dp[0] = true;
    
    for (int num : nums) {
        for (int j = target; j >= num; j--) {
            dp[j] = dp[j] || dp[j - num];
        }
    }
    
    return dp[target];
}
```

    - **Java模板**:
    - **难点**: 空间优化到一维
    - **扩展**: 分割成k个相等子集

```java
public int uniquePaths(int m, int n) {
    int[][] dp = new int[m][n];
    
    // 初始化第一行和第一列
    for (int i = 0; i < m; i++) dp[i][0] = 1;
    for (int j = 0; j < n; j++) dp[0][j] = 1;
    
    for (int i = 1; i < m; i++) {
        for (int j = 1; j < n; j++) {
            dp[i][j] = dp[i-1][j] + dp[i][j-1];
        }
    }
    
    return dp[m-1][n-1];
}

// 空间优化版本
public int uniquePathsOptimized(int m, int n) {
    int[] dp = new int[n];
    Arrays.fill(dp, 1);
    
    for (int i = 1; i < m; i++) {
        for (int j = 1; j < n; j++) {
            dp[j] = dp[j] + dp[j-1];
        }
    }
    
    return dp[n-1];
}
```

```java
public int minPathSum(int[][] grid) {
    int m = grid.length, n = grid[0].length;
    int[][] dp = new int[m][n];
    
    dp[0][0] = grid[0][0];
    
    // 初始化第一行
    for (int j = 1; j < n; j++) {
        dp[0][j] = dp[0][j-1] + grid[0][j];
    }
    
    // 初始化第一列
    for (int i = 1; i < m; i++) {
        dp[i][0] = dp[i-1][0] + grid[i][0];
    }
    
    for (int i = 1; i < m; i++) {
        for (int j = 1; j < n; j++) {
            dp[i][j] = Math.min(dp[i-1][j], dp[i][j-1]) + grid[i][j];
        }
    }
    
    return dp[m-1][n-1];
}

// 空间优化版本
public int minPathSumOptimized(int[][] grid) {
    int m = grid.length, n = grid[0].length;
    int[] dp = new int[n];
    
    dp[0] = grid[0][0];
    for (int j = 1; j < n; j++) {
        dp[j] = dp[j-1] + grid[0][j];
    }
    
    for (int i = 1; i < m; i++) {
        dp[0] += grid[i][0];
        for (int j = 1; j < n; j++) {
            dp[j] = Math.min(dp[j], dp[j-1]) + grid[i][j];
        }
    }
    
    return dp[n-1];
}
```

```java
public int maximalSquare(char[][] matrix) {
    int m = matrix.length, n = matrix[0].length;
    int[][] dp = new int[m + 1][n + 1];
    int maxSide = 0;
    
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (matrix[i-1][j-1] == '1') {
                dp[i][j] = Math.min(Math.min(dp[i-1][j], dp[i][j-1]), dp[i-1][j-1]) + 1;
                maxSide = Math.max(maxSide, dp[i][j]);
            }
        }
    }
    
    return maxSide * maxSide;
}

// 空间优化版本
public int maximalSquareOptimized(char[][] matrix) {
    int m = matrix.length, n = matrix[0].length;
    int[] dp = new int[n + 1];
    int maxSide = 0, prev = 0;
    
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            int temp = dp[j];
            if (matrix[i-1][j-1] == '1') {
                dp[j] = Math.min(Math.min(dp[j], dp[j-1]), prev) + 1;
                maxSide = Math.max(maxSide, dp[j]);
            } else {
                dp[j] = 0;
            }
            prev = temp;
        }
    }
    
    return maxSide * maxSide;
}
```

    - **Java模板**:
    - **难点**: 理解正方形的递推关系
    - **扩展**: 最大矩形、最大子矩阵

```java
public int maxCoins(int[] nums) {
    int n = nums.length;
    int[] newNums = new int[n + 2];
    newNums[0] = newNums[n + 1] = 1;
    for (int i = 0; i < n; i++) {
        newNums[i + 1] = nums[i];
    }
    
    int[][] dp = new int[n + 2][n + 2];
    
    for (int len = 3; len <= n + 2; len++) {
        for (int i = 0; i <= n + 2 - len; i++) {
            int j = i + len - 1;
            for (int k = i + 1; k < j; k++) {
                dp[i][j] = Math.max(dp[i][j], 
                    dp[i][k] + dp[k][j] + newNums[i] * newNums[k] * newNums[j]);
            }
        }
    }
    
    return dp[0][n + 1];
}
```

    - **Java模板**:
    - **难点**: 区间的划分和合并策略

## 💡 Java解题技巧总结
### 🔍 状态定义技巧
+ **明确含义**: 状态必须有明确的数学含义
+ **无后效性**: 当前状态不依赖于未来状态
+ **最优子结构**: 问题的最优解包含子问题的最优解

### 🎯 转移方程技巧
+ **枚举决策**: 考虑所有可能的决策
+ **边界处理**: 特别注意数组越界
+ **状态压缩**: 降低空间复杂度

### ⚡ Java优化策略
+ **滚动数组**: 二维DP优化为一维
+ **单调性**: 利用单调队列/栈优化
+ **数学性质**: 利用数学公式直接求解
+ **位运算**: 状态压缩DP中的位操作

### 🛡️ Java编程注意事项
+ **数组初始化**: 使用`Arrays.fill()`或循环初始化
+ **边界检查**: 防止数组越界异常
+ **整数溢出**: 使用`long`类型或取模运算
+ **空间复杂度**: 优先考虑空间优化版本
