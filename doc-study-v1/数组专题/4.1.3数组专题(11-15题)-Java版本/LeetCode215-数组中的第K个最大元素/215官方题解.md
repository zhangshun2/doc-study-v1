# 215官方题解

```java
class Solution {
    int quickselect(int[] nums, int l, int r, int k) {
        if (l == r) return nums[k];
        int x = nums[l], i = l - 1, j = r + 1;
        while (i < j) {
            do i++; while (nums[i] < x);
            do j--; while (nums[j] > x);
            if (i < j){
                int tmp = nums[i];
                nums[i] = nums[j];
                nums[j] = tmp;
            }
        }
        if (k <= j) return quickselect(nums, l, j, k);
        else return quickselect(nums, j + 1, r, k);
    }
    public int findKthLargest(int[] _nums, int k) {
        int n = _nums.length;
        return quickselect(_nums, 0, n - 1, n - k);
    }
}
```

## 阿里开发规范版本
```java
/**
 * LeetCode 215: 数组中的第K个最大元素
 * 使用快速选择算法实现，平均时间复杂度 O(n)
 * 
 * @author 开发者
 * @since 1.0
 */
public class KthLargestElementSolution {
    
    /**
     * 查找数组中第K个最大元素
     * 
     * @param nums 输入数组，不能为null
     * @param k 第k大的元素，范围[1, nums.length]
     * @return 第k大的元素值
     * @throws IllegalArgumentException 当输入参数不合法时抛出
     */
    public int findKthLargest(int[] nums, int k) {
        // 参数校验
        if (nums == null || nums.length == 0) {
            throw new IllegalArgumentException("数组不能为空");
        }
        if (k <= 0 || k > nums.length) {
            throw new IllegalArgumentException("k值必须在[1, " + nums.length + "]范围内");
        }
        
        // 第k大元素等价于第(n-k)小元素（0-based索引）
        int targetIndex = nums.length - k;
        return quickSelect(nums, 0, nums.length - 1, targetIndex);
    }
    
    /**
     * 快速选择算法核心实现
     * 基于快速排序的分区思想，每次只处理包含目标元素的一半
     * 
     * @param nums 待处理数组
     * @param leftBound 左边界（包含）
     * @param rightBound 右边界（包含）
     * @param targetIndex 目标元素的索引位置
     * @return 目标位置的元素值
     */
    private int quickSelect(int[] nums, int leftBound, int rightBound, int targetIndex) {
        // 递归终止条件：搜索范围缩小到单个元素
        if (leftBound == rightBound) {
            return nums[targetIndex];
        }
        
        // 执行分区操作，返回基准元素的最终位置
        int pivotIndex = partition(nums, leftBound, rightBound);
        
        // 根据基准位置决定下一步搜索方向
        if (targetIndex == pivotIndex) {
            // 找到目标元素
            return nums[targetIndex];
        } else if (targetIndex < pivotIndex) {
            // 目标在左半部分，递归搜索左边
            return quickSelect(nums, leftBound, pivotIndex - 1, targetIndex);
        } else {
            // 目标在右半部分，递归搜索右边
            return quickSelect(nums, pivotIndex + 1, rightBound, targetIndex);
        }
    }
    
    /**
     * 分区操作：将数组分为两部分
     * 左半部分：所有元素 <= 基准值
     * 右半部分：所有元素 >= 基准值
     * 
     * @param nums 待分区数组
     * @param leftBound 左边界（包含）
     * @param rightBound 右边界（包含）
     * @return 基准元素的最终位置索引
     */
    private int partition(int[] nums, int leftBound, int rightBound) {
        // 选择最左元素作为基准值
        int pivotValue = nums[leftBound];
        
        // 初始化双指针（从边界外开始，确保do-while至少执行一次）
        int leftPointer = leftBound - 1;
        int rightPointer = rightBound + 1;
        
        // 双指针向中间移动，交换不符合条件的元素
        while (leftPointer < rightPointer) {
            // 从左向右找第一个 >= 基准值的元素
            do {
                leftPointer++;
            } while (nums[leftPointer] < pivotValue);
            
            // 从右向左找第一个 <= 基准值的元素
            do {
                rightPointer--;
            } while (nums[rightPointer] > pivotValue);
            
            // 如果两指针未相遇，交换元素
            if (leftPointer < rightPointer) {
                swapElements(nums, leftPointer, rightPointer);
            }
        }
        
        // 返回分区点（右指针的最终位置）
        return rightPointer;
    }
    
    /**
     * 交换数组中两个位置的元素
     * 
     * @param nums 数组
     * @param indexA 位置A
     * @param indexB 位置B
     */
    private void swapElements(int[] nums, int indexA, int indexB) {
        int temp = nums[indexA];
        nums[indexA] = nums[indexB];
        nums[indexB] = temp;
    }
}
```
